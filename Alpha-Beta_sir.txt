import math

leaves = [3, 9, 1, 8, 5, 2, 7, 4, 6]

def alpha_beta(depth, index, is_max, alpha, beta):
    if depth == 2:
        start = index * 3
        return min(leaves[start : start + 3])

    if is_max:
        val = -math.inf
        for i in range(3):
            res = alpha_beta(depth + 1, i, False, alpha, beta)
            val = max(val, res)
            alpha = max(alpha, val)
            if beta <= alpha:
                break
        return val
    else:
        val = math.inf
        # Yahan index range 1 rakhi hai kyunke 9 leaves depth 2 par cover ho rahi hain
        iterations = 3 if depth == 0 else 1 
        for i in range(iterations):
            res = alpha_beta(depth + 1, i, True, alpha, beta)
            val = min(val, res)
            beta = min(beta, val)
            if beta <= alpha:
                break
        return val

result = alpha_beta(0, 0, False, -math.inf, math.inf)
print(result)
