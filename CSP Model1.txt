# OR-Tools: used for constraint programming (CSP problems)
from ortools.sat.python import cp_model


model = cp_model.CpModel()


a = model.NewIntVar(1, 3, 'a')  # Create variable a
b = model.NewIntVar(1, 3, 'b')  # Create variable b
c = model.NewIntVar(1, 3, 'c')  # Create variable c


model.Add(a != b)
model.Add(b != c)
model.Add(a != c)


solver = cp_model.CpSolver()


solution_printer = cp_model.VarArraySolutionPrinter([a, b, c])

solver.parameters.enumerate_all_solutions = True


solver.Solve(model, solution_printer)

print("OPTIMAL:", cp_model.OPTIMAL)
print("FEASIBLE:", cp_model.FEASIBLE)
print("INFEASIBLE:", cp_model.INFEASIBLE)
print("UNKNOWN:", cp_model.UNKNOWN)
