import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, r2_score


# =========================
# GENERATE SAMPLE DATA
# =========================
np.random.seed(42)

# Feature (X)
x = np.random.rand(100, 1) * 10  # values between 0 and 10

# Target (y) with noise
y = 3 * x.flatten() + np.random.randn(100) * 2


# =========================
# TRAIN-TEST SPLIT
# =========================
x_train, x_test, y_train, y_test = train_test_split(
    x, y, test_size=0.2, random_state=42
)


# =========================
# MODEL TRAINING
# =========================
LR = LinearRegression()
model = LR.fit(x_train, y_train)


# =========================
# PREDICTIONS
# =========================
y_pred = model.predict(x_test)


# =========================
# MODEL EVALUATION
# =========================

# Mean Squared Error
mse = mean_squared_error(y_test, y_pred)

# R2 Score (Testing Accuracy)
test_accuracy = r2_score(y_test, y_pred)

# R2 Score (Training Accuracy)
train_accuracy = r2_score(y_train, model.predict(x_train))


# =========================
# OUTPUT
# =========================
print("Predictions:", y_pred)
print("\nMean Squared Error:", mse)

print("\nTraining Accuracy (R2 Score):", train_accuracy)
print("Testing Accuracy (R2 Score):", test_accuracy)
